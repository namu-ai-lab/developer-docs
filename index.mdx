```python Python
from namu import Namu

client = Namu()  # reads NAMU_API_KEY

audio = client.speech("Sannu da zuwa!", voice="nadia")
open("welcome.wav", "wb").write(audio)
```

```javascript JavaScript
import { writeFile } from "node:fs/promises";
import { Namu } from "namu-ai";

const namu = new Namu(); // reads NAMU_API_KEY

const audio = await namu.speech("Sannu da zuwa!", { voice: "nadia" });
await writeFile("welcome.wav", Buffer.from(audio));
```

```bash cURL
curl https://api.namuai-lab.com/v1/speech \
  -H "Authorization: Bearer $NAMU_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"text": "Sannu da zuwa!", "voice": "nadia"}' \
  -o welcome.wav
```
